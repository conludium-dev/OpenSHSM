cmake_minimum_required(VERSION 3.20)
project(OpenSHSMStockToolchain LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
file(READ "${NWII_STOCK_ROOT}/preparation.json" preparation)
foreach(dependency IN ITEMS tomlplusplus sdl2 glad)
    string(JSON archive GET "${preparation}" dependencies ${dependency} path)
    string(JSON digest GET "${preparation}" dependencies ${dependency} sha256)
    FetchContent_Declare(${dependency} URL "${archive}" URL_HASH "SHA256=${digest}")
endforeach()
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
set(GLAD_PROFILE "core" CACHE STRING "" FORCE)
set(GLAD_API "gl=3.3" CACHE STRING "" FORCE)
set(GLAD_GENERATOR "c" CACHE STRING "" FORCE)
set(GLAD_REPRODUCIBLE ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(tomlplusplus sdl2 glad)
add_subdirectory("${NWII_STOCK_ROOT}/upstream-source/nWiiAnalyzer" nWiiAnalyzer)
add_subdirectory("${NWII_STOCK_ROOT}/upstream-source/nWiiRuntime" nWiiRuntime)
add_subdirectory("${NWII_STOCK_ROOT}/upstream-source/nWiiRecomp" nWiiRecomp)
